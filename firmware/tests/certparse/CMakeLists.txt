# certparse conformance (step 6a, docs/roadmap/wolfssl-migration.md).
#
# Covers the X.509 reader seam, and in particular certificate *chain*
# verification — which had no test at all before this suite. That gap matters
# more here than under mbedTLS: a WOLFCRYPT_ONLY build has no library chain
# verifier to call (wolfSSL's lives in the TLS layer), so src/ca/
# certparse_wolfssl.c walks the path itself. Hand-written path building is
# where the host seam shipped a trust-anchor bug; see src/main.c for why the
# negative cases carry the weight.
#
# The fixtures come from libcantil/tests, generated by the openssl CLI — a
# third implementation, which is what keeps this from being wolfCrypt grading
# its own homework. Certificates Cantil *generates* are checked by
# scripts/openssl_gate.sh instead.
#
# Build/run: ./scripts/test.sh certparse

cmake_minimum_required(VERSION 3.20.0)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(certparse_tests LANGUAGES C)

set(FW ${CMAKE_CURRENT_SOURCE_DIR}/../..)

target_sources(app PRIVATE
    src/main.c
    ${FW}/src/ca/certparse_wolfssl.c
    # wolfCrypt's RNG seam. Nothing here needs entropy, but wc_RNG_GenerateBlock
    # is reachable from the ASN.1 code and must resolve.
    ${FW}/src/crypto/wc_rand.c
)

target_include_directories(app PRIVATE
    ${FW}/src
    ${FW}/include
    # Fixtures are shared with the host seam's tests, so both sides of the
    # wire are checked against the same certificates.
    ${FW}/../libcantil/tests
)

# cantil_wolfssl_user_settings.h lives here.
zephyr_include_directories(${FW}/include)

# Carries WOLFSSL_USER_SETTINGS / WOLFSSL_ZEPHYR defines.
target_link_libraries(app PRIVATE wolfSSL)
